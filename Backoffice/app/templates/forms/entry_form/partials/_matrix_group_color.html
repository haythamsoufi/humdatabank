{# Inline style for a matrix column-group header. Empty when unset or the default grey (#f3f4f6), so the header keeps the same classes as a normal column. Custom colours set background and a contrasting text colour. Only #rrggbb values are emitted. #} {% macro matrix_group_header_style(group_meta) -%} {%- set ns = namespace(raw='', valid=false) -%} {%- if group_meta is mapping -%} {%- set candidate = group_meta.get('header_color') -%} {%- if candidate is string -%} {%- set ns.raw = candidate|trim|lower -%} {%- endif -%} {%- endif -%} {%- set ns.valid = (ns.raw|length == 7 and ns.raw[0] == '#') -%} {%- if ns.valid -%} {%- for ch in ns.raw[1:] -%} {%- if ch not in '0123456789abcdef' -%}{%- set ns.valid = false -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- if ns.valid and ns.raw != '#f3f4f6' -%} {%- set r = ns.raw[1:3]|int(base=16) -%} {%- set g = ns.raw[3:5]|int(base=16) -%} {%- set b = ns.raw[5:7]|int(base=16) -%} {%- set lum = (r * 0.299) + (g * 0.587) + (b * 0.114) -%} {%- set text = '#4b5563' if lum >= 160 else '#ffffff' -%} style="background-color: {{ ns.raw }}; color: {{ text }};" {%- endif -%} {%- endmacro %}
